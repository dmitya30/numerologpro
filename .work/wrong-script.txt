cd ~/projects/numerologpro

set +e
set +u
set +o pipefail 2>/dev/null

expected="922ca7b21d4e10c76ba5f9d3b581bc0f8d25188b"
actual="$(git rev-parse HEAD 2>/dev/null)"
ok=1

pages="elyor/index.html quantocode/index.html oracle/index.html"
tmp_root="${TMPDIR:-/tmp}/numerologpro-home-redesign"
analytics_scripts="$tmp_root/analytics-scripts.html"
analytics_noscript="$tmp_root/analytics-noscript.html"

printf 'Expected HEAD: %s\n' "$expected"
printf 'Actual HEAD:   %s\n' "$actual"

if [ "$actual" != "$expected" ]; then
    printf 'FAIL: неожиданный HEAD.\n'
    ok=0
fi

if [ "$ok" -eq 1 ]; then
    git fetch origin main
    remote="$(git rev-parse origin/main 2>/dev/null)"
    printf 'origin/main:   %s\n' "$remote"

    if [ "$remote" != "$expected" ]; then
        printf 'FAIL: origin/main не совпадает с ожидаемым HEAD.\n'
        ok=0
    fi
fi

if [ "$ok" -eq 1 ]; then
    if [ -n "$(git diff --cached --name-only)" ] || [ -n "$(git diff --name-only)" ]; then
        printf 'FAIL: обнаружены staged или изменённые tracked-файлы:\n'
        git status --short
        ok=0
    fi
fi

if [ "$ok" -eq 1 ]; then
    untracked="$(git ls-files --others --exclude-standard | sort)"
    expected_untracked="$(printf '%s\n' \
        'assets/images/home/hero-desktop.webp' \
        'assets/images/home/hero-mobile.webp' \
        'assets/images/home/og-quiet-house.jpg' \
        | sort)"

    if [ "$untracked" != "$expected_untracked" ]; then
        printf 'FAIL: набор untracked-файлов отличается от ожидаемых изображений:\n%s\n' "$untracked"
        ok=0
    else
        printf 'PASS: подтверждены только три новых изображения.\n'
    fi
fi

if [ "$ok" -eq 1 ]; then
    desktop_info="$(file assets/images/home/hero-desktop.webp)"
    mobile_info="$(file assets/images/home/hero-mobile.webp)"
    og_info="$(file assets/images/home/og-quiet-house.jpg)"

    printf '%s\n%s\n%s\n' "$desktop_info" "$mobile_info" "$og_info"

    if ! printf '%s' "$desktop_info" | grep -q 'WebP image.*1920x1200'; then
        printf 'FAIL: desktop hero имеет неожиданный формат или размер.\n'
        ok=0
    fi

    if ! printf '%s' "$mobile_info" | grep -q 'WebP image.*1080x1360'; then
        printf 'FAIL: mobile hero имеет неожиданный формат или размер.\n'
        ok=0
    fi

    if ! printf '%s' "$og_info" | grep -q 'JPEG image data.*1200x630'; then
        printf 'FAIL: OG имеет неожиданный формат или размер.\n'
        ok=0
    fi
fi

if [ "$ok" -eq 1 ]; then
    mkdir -p "$tmp_root"
    rm -f "$analytics_scripts" "$analytics_noscript"

    ga_before="$(grep -o 'googletagmanager\.com' index.html | wc -l | tr -d ' ')"
    ym_before="$(grep -o 'mc\.yandex\.ru' index.html | wc -l | tr -d ' ')"

    awk '
        function flush_script() {
            if (buffer ~ /googletagmanager\.com|mc\.yandex\.ru/) {
                print buffer
            }
            buffer = ""
            inside = 0
        }
        {
            if (!inside && $0 ~ /<script/) {
                inside = 1
                buffer = $0
                if ($0 ~ /<\/script>/) {
                    flush_script()
                }
                next
            }
            if (inside) {
                buffer = buffer ORS $0
                if ($0 ~ /<\/script>/) {
                    flush_script()
                }
            }
        }
    ' index.html > "$analytics_scripts"

    awk '
        function flush_noscript() {
            if (buffer ~ /mc\.yandex\.ru/) {
                print buffer
            }
            buffer = ""
            inside = 0
        }
        {
            if (!inside && $0 ~ /<noscript/) {
                inside = 1
                buffer = $0
                if ($0 ~ /<\/noscript>/) {
                    flush_noscript()
                }
                next
            }
            if (inside) {
                buffer = buffer ORS $0
                if ($0 ~ /<\/noscript>/) {
                    flush_noscript()
                }
            }
        }
    ' index.html > "$analytics_noscript"

    ga_saved="$(grep -o 'googletagmanager\.com' "$analytics_scripts" | wc -l | tr -d ' ')"
    ym_saved="$(
        {
            grep -o 'mc\.yandex\.ru' "$analytics_scripts"
            grep -o 'mc\.yandex\.ru' "$analytics_noscript"
        } | wc -l | tr -d ' '
    )"

    printf 'Analytics before: GA=%s YM=%s\n' "$ga_before" "$ym_before"
    printf 'Analytics saved:  GA=%s YM=%s\n' "$ga_saved" "$ym_saved"

    if [ "$ga_before" -ne 4 ] || [ "$ym_before" -ne 8 ] || \
       [ "$ga_saved" -ne "$ga_before" ] || [ "$ym_saved" -ne "$ym_before" ]; then
        printf 'FAIL: точное сохранение аналитики не подтверждено.\n'
        ok=0
    else
        printf 'PASS: аналитика извлечена без изменения маркеров.\n'
    fi
fi

if [ "$ok" -eq 1 ]; then
    before_hashes="$tmp_root/product-pages-before.sha256"
    sha256sum $pages > "$before_hashes"

    mkdir -p assets/css assets/js
fi

        code="$(curl -sS -o /dev/null -w '%{http_code}' "http://127.0.0.1:8765$route")"
        printf '%s %s\n' "$code" "$route"

        if [ "$code" != "200" ]; then
            ok=0
        fi
    done

    kill "$server_pid" 2>/dev/null
    wait "$server_pid" 2>/dev/null

    if [ "$ok" -eq 1 ]; then
        printf 'PASS: все обязательные локальные маршруты и ресурсы возвращают HTTP 200.\n'
    else
        printf 'FAIL: локальная HTTP-проверка не прошла.\n'
    fi
fi

if [ "$ok" -eq 1 ]; then
    git diff --check

    if [ "$?" -ne 0 ]; then
        printf 'FAIL: git diff --check обнаружил ошибки.\n'
        ok=0
    else
        printf 'PASS: git diff --check.\n'
    fi
fi

if [ "$ok" -eq 1 ]; then
    git add -A

    if [ -n "$(git diff --name-only)" ] || [ -n "$(git ls-files --others --exclude-standard)" ]; then
        printf 'FAIL: после staging остались unstaged/untracked файлы.\n'
        ok=0
    fi
fi

if [ "$ok" -eq 1 ]; then
    if git diff --cached -U0 | grep -Eiq \
        'BEGIN (RSA |EC |OPENSSH )?PRIVATE KEY|authorization:[[:space:]]*bearer|password[[:space:]]*[:=][[:space:]]*[^[:space:]]+|api[_-]?key[[:space:]]*[:=][[:space:]]*[^[:space:]]+'
    then
        printf 'FAIL: staged diff содержит возможный секрет.\n'
        ok=0
    else
        printf 'PASS: очевидные секреты в staged diff не обнаружены.\n'
    fi
fi

printf '\nStaged diff stat:\n'
git diff --cached --stat

printf '\nStaged files:\n'
git diff --cached --name-status

if [ "$ok" -eq 1 ]; then
    git diff --cached --check

    if [ "$?" -ne 0 ]; then
        printf 'FAIL: staged whitespace-проверка не прошла.\n'
        ok=0
    fi
fi

if [ "$ok" -eq 1 ]; then
    git commit -m "feat: create Quiet House homepage and custom 404"

    if [ "$?" -ne 0 ]; then
        printf 'FAIL: commit не создан.\n'
        ok=0
    fi
fi

if [ "$ok" -eq 1 ]; then
    git push origin main

    if [ "$?" -ne 0 ]; then
        printf 'FAIL: push не выполнен.\n'
        ok=0
    fi
fi

if [ "$ok" -eq 1 ]; then
    local_head="$(git rev-parse HEAD)"
    remote_head="$(git ls-remote origin refs/heads/main | awk '{print $1}')"

    printf 'Local HEAD:  %s\n' "$local_head"
    printf 'Remote HEAD: %s\n' "$remote_head"

    if [ "$local_head" != "$remote_head" ]; then
        printf 'FAIL: локальный и удалённый HEAD не совпадают.\n'
        ok=0
    fi
fi

printf '\nFinal status:\n'
git status --short

if [ "$ok" -eq 1 ] && [ -z "$(git status --short)" ]; then
    printf '\nPASS: новая главная и 404 созданы, проверены, закоммичены и отправлены в origin/main.\n'
    printf 'STOP-GATE: после обновления GitHub Pages требуется человеческий desktop/mobile QA.\n'
else
    printf '\nFAIL: операция завершена не полностью; смотри первое сообщение FAIL выше.\n'
fi
